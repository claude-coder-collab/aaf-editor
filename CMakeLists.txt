cmake_minimum_required(VERSION 3.28)

project(aaf_editor VERSION 0.1.0 DESCRIPTION "AAF library and editor" LANGUAGES CXX)

if(MSVC)
    set(AAF_CXX_STANDARD 23)
else()
    set(AAF_CXX_STANDARD 26)
endif()
set(CMAKE_CXX_STANDARD ${AAF_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

include(cmake/ProjectOptions.cmake)
include(cmake/Dependencies.cmake)

add_subdirectory(libs/common)
add_subdirectory(libs/cfb)
add_subdirectory(libs/core)
add_subdirectory(libs/edit)
add_subdirectory(libs/timeline)
add_subdirectory(apps/rpc)
add_subdirectory(apps/aaftool)
if(AAF_BUILD_EDITOR)
    add_subdirectory(apps/editor)
endif()

if(AAF_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

if(AAF_BUILD_FUZZERS)
    add_subdirectory(tests/fuzz)
endif()
